## Migration 021

```sql
-- Per-user machines (image build state, manifest, image digest)
CREATE TABLE machines (
  id TEXT PRIMARY KEY,                          -- e.g. 'mach_<userid>_<n>' or 'default-scruple-canvas-v1'
  user_id TEXT,                                 -- NULL for shared default
  label TEXT NOT NULL,
  manifest_json TEXT NOT NULL,                  -- canonical JSON
  manifest_hash TEXT NOT NULL,                  -- sha256 hex of manifest_json (canonicalized)
  build_status TEXT NOT NULL DEFAULT 'pending', -- 'pending' | 'building' | 'ready' | 'failed'
  build_error TEXT,
  modal_image_digest TEXT,                      -- Modal's image sha after build
  created_at INTEGER NOT NULL,
  ready_at INTEGER,
  archived_at INTEGER                            -- non-null = retired (still readable for receipts)
);
CREATE INDEX idx_machines_user ON machines(user_id);
CREATE INDEX idx_machines_hash ON machines(manifest_hash);

-- Time-series snapshots — when user edits manifest, NEW machines row + machine_versions audit row
CREATE TABLE machine_versions (
  id INTEGER PRIMARY KEY,
  machine_id TEXT NOT NULL,                      -- FK -> machines.id
  manifest_hash TEXT NOT NULL,
  manifest_json TEXT NOT NULL,
  created_at INTEGER NOT NULL
);
CREATE INDEX idx_mv_machine ON machine_versions(machine_id);

-- Leaf v2.2 — store manifest_hash on iterations (preimage component)
ALTER TABLE iterations ADD COLUMN machine_manifest_hash TEXT;

-- Leaf scheme bump
ALTER TABLE iterations ADD COLUMN leaf_scheme TEXT NOT NULL DEFAULT 'v2.1';

-- Publication mode (WO-9 will reference this; lands now to avoid two migrations)
ALTER TABLE iterations ADD COLUMN workflow_publication TEXT NOT NULL DEFAULT 'full';

-- Witnesses table: same machine_manifest_hash column for canonical record verification
ALTER TABLE witnesses ADD COLUMN machine_manifest_hash TEXT;

-- Seed the shared default machine
INSERT INTO machines (id, user_id, label, manifest_json, manifest_hash, build_status, created_at)
VALUES ('default-scruple-canvas-v1', NULL, 'Default Scruple Canvas v1',
        '<MANIFEST_JSON_FROM_CONFIG>',  -- migration runner reads config/default-machine-manifest.json
        '<HASH_OF_MANIFEST>',
        'pending',                       -- WO-7 build worker flips to 'ready' after first build
        unixepoch());
```
